import { GameState, Player } from '@imposter/shared';

export const getViewer = (gameState: GameState, playerId: string): Player | undefined =>
  gameState.players.find((player) => player.id === playerId);

export const isHost = (gameState: GameState, playerId: string): boolean =>
  gameState.hostPlayerId === playerId;

export const alivePlayers = (gameState: GameState): Player[] =>
  gameState.players.filter((player) => player.isAlive);

      - Returns false for Winner.NONE regardless of role
      - Returns false when role does not match winning side

    getRoleLabel(role: Role | null): string
      - Returns 'CITIZEN' for Role.CITIZEN
      - Returns 'SPY' for Role.SPY
      - Returns 'WHITE ROLE' for Role.WHITE

      - NOTE: returns a plain string constant — NOT a t() call. Components call t() themselves.

    getRoleColorClass(role: Role | null): string
      - Returns 'text-blue-400' for Role.CITIZEN
      - Returns 'text-red-400' for Role.SPY
      - Returns 'text-neutral-400' for Role.WHITE
      - Returns 'text-neutral-400' for null or unrecognised value

    isGameOver(gameState: GameState): boolean
      - Returns true if gameState.phase === Phase.GAME_ENDED
      - Returns false for all other phases

    canViewerSeeWord(viewer: Player | undefined, gameState: GameState): boolean
      - Returns true if viewer?.word is non-null AND game is NOT over
      - Returns false if viewer is undefined
      - Returns false if viewer.word is null or undefined
      - Returns false if isGameOver(gameState) is true
